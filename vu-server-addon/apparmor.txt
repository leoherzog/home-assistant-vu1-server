#include <tunables/global>

profile vu-server-addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Broad file and network access is intentional. A custom profile replaces Docker's default, so its deny rules are copied below.
  file,
  network,
  signal (send) set=(kill,term,int,hup,cont),
  # Lets the container runtime deliver SIGTERM on stop.
  signal (receive) peer=runc,
  signal (receive) peer=crun,
  # Capabilities nginx needs.
  capability setuid,
  capability setgid,
  capability chown,
  capability dac_override,

  # Docker's default-profile denies.
  deny network alg,
  deny network vsock,
  deny @{PROC}/* w,
  deny @{PROC}/{[^1-9/],[^1-9/][^0-9/],[^1-9s/][^0-9y/][^0-9s/],[^1-9/][^0-9/][^0-9/][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,
  deny mount,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,
}
